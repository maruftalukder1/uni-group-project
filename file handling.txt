
#include <iostream>
#include <fstream>
#include <string>
#include <vector>
using namespace std;

namespace MeghaFileHandling
{
    bool saveRecord(const string& fileName, const string& data)
    {
        ofstream file(fileName, ios::app);

        if (!file)
        {
            cout << "Error opening file.\n";
            return false;
        }

        file << data << '\n';
        file.close();

        cout << "Record saved successfully.\n";
        return true;
    }

    void viewRecords(const string& fileName)
    {
        ifstream file(fileName);
        string record;
        bool found = false;

        if (!file)
        {
            cout << "No records found.\n";
            return;
        }

        cout << "\n--- Saved Records ---\n";

        while (getline(file, record))
        {
            cout << record << '\n';
            found = true;
        }

        if (!found)
            cout << "No records available.\n";
    }

    bool searchRecords(const string& fileName, const string& keyword)
    {
        ifstream file(fileName);
        string record;
        bool found = false;

        if (!file)
        {
            cout << "File not found.\n";
            return false;
        }

        while (getline(file, record))
        {
            if (record.find(keyword) != string::npos)
            {
                cout << "Record found: " << record << '\n';
                found = true;
            }
        }

        if (!found)
            cout << "No matching record found.\n";

        return found;
    }

    bool updateRecord(const string& fileName,
                      const string& oldRecord,
                      const string& newRecord)
    {
        ifstream file(fileName);
        if (!file)
        {
            cout << "File not found.\n";
            return false;
        }

        vector<string> records;
        string record;
        bool found = false;

        while (getline(file, record))
        {
            if (record == oldRecord && !found)
            {
                records.push_back(newRecord);
                found = true;
            }
            else
                records.push_back(record);
        }
        file.close();

        if (!found)
        {
            cout << "Record not found.\n";
            return false;
        }

        ofstream out(fileName);
        if (!out)
        {
            cout << "Error updating file.\n";
            return false;
        }

        for (const string& item : records)
            out << item << '\n';

        cout << "Record updated successfully.\n";
        return true;
    }

    bool deleteRecord(const string& fileName, const string& targetRecord)
    {
        ifstream file(fileName);
        if (!file)
        {
            cout << "File not found.\n";
            return false;
        }

        vector<string> records;
        string record;
        bool found = false;

        while (getline(file, record))
        {
            if (record == targetRecord && !found)
                found = true;
            else
                records.push_back(record);
        }
        file.close();

        if (!found)
        {
            cout << "Record not found.\n";
            return false;
        }

        ofstream out(fileName);
        if (!out)
        {
            cout << "Error updating file.\n";
            return false;
        }

        for (const string& item : records)
            out << item << '\n';

        cout << "Record deleted successfully.\n";
        return true;
    }
}

int main()
{
    int choice;
    string data, keyword, oldRecord, newRecord;

    do
    {
        cout << "\n=== MEGHA: FILE HANDLING ===\n";
        cout << "1. Save Record\n";
        cout << "2. View Records\n";
        cout << "3. Search Record\n";
        cout << "4. Update Record\n";
        cout << "5. Delete Record\n";
        cout << "0. Exit\n";
        cout << "Enter choice: ";
        cin >> choice;
        cin.ignore(10000, '\n');

        switch (choice)
        {
            case 1:
                cout << "Enter record: ";
                getline(cin, data);
                MeghaFileHandling::saveRecord("pets.txt", data);
                break;

            case 2:
                MeghaFileHandling::viewRecords("pets.txt");
                break;

            case 3:
                cout << "Enter search keyword: ";
                getline(cin, keyword);
                MeghaFileHandling::searchRecords("pets.txt", keyword);
                break;

            case 4:
                cout << "Enter exact old record: ";
                getline(cin, oldRecord);
                cout << "Enter new record: ";
                getline(cin, newRecord);
                MeghaFileHandling::updateRecord(
                    "pets.txt", oldRecord, newRecord);
                break;

            case 5:
                cout << "Enter exact record to delete: ";
                getline(cin, oldRecord);
                MeghaFileHandling::deleteRecord("pets.txt", oldRecord);
                break;

            case 0:
                cout << "Exiting File Handling Module.\n";
                break;

            default:
                cout << "Invalid choice.\n";
        }
    }
    while (choice != 0);

    return 0;
}
